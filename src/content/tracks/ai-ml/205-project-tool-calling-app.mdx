## المشكلة

**بناء مساعد بسيط** يقدر:

1. البحث عن تعريف كلمة (قاموس).
2. جلب حالة الطقس لمدينة.
3. إجراء عمليات حسابية.

كل أداة لها **JSON Schema** صارم. النموذج يقرّر أيّ أداة يستدعي ومتى، أنت تنفّذ.

## 2. تنفيذ الأدوات

```python
DICTIONARY = {
    "ذكاء":   "قدرة على الفهم، التعلّم، والاستنتاج.",
    "خوارزمية": "خطوات محدودة لحلّ مشكلة أو تنفيذ مهمة.",
    "بيانات":  "حقائق أو معلومات يمكن معالجتها.",
    "تعلّم آلي": "فرع من الذكاء الاصطناعي يتعلّم من البيانات.",
}

WEATHER = {
    "عمّان":    {"temp_c": 25, "condition": "مشمس"},
    "القاهرة":   {"temp_c": 32, "condition": "غائم جزئيًا"},
    "الرياض":    {"temp_c": 40, "condition": "حار"},
    "دبي":      {"temp_c": 38, "condition": "رطب"},
}


def lookup_word(word: str) -> str:
    if word not in DICTIONARY:
        return json.dumps({"error": f"الكلمة '{word}' غير موجودة في القاموس."})
    return json.dumps({"word": word, "definition": DICTIONARY[word]}, ensure_ascii=False)


def get_weather(city: str) -> str:
    if city not in WEATHER:
        return json.dumps({"error": f"لا توجد بيانات لمدينة {city}."})
    info = WEATHER[city]
    return json.dumps(info, ensure_ascii=False)


def calculate(expression: str) -> str:
    """تنفيذ تعبير رياضي آمن (لا eval على مدخلات خام)."""
    # ✅ whitelist للعمليات والدوال المسموحة
    allowed_names = {
        "sqrt": math.sqrt,
        "sin": math.sin,
        "cos": math.cos,
        "tan": math.tan,
        "pi": math.pi,
        "e": math.e,
        "abs": abs,
        "round": round,
    }

    try:
        # ✅ compile ثم eval بسياق محدود
        code = compile(expression, "<user_expr>", "eval")
        # التحقق من أنّ التعبير يستخدم أسماء مسموحة فقط
        for name in code.co_names:
            if name not in allowed_names:
                return json.dumps({"error": f"اسم غير مسموح: {name}"})
        result = eval(code, {"__builtins__": {}}, allowed_names)
        return json.dumps({"expression": expression, "result": result})
    except Exception as e:
        return json.dumps({"error": str(e)})
```

**⚠️ ملاحظة أمان**: `calculate()` يستخدم `compile()` و eval بسياق محدود. **لا تستخدم `eval(input)` مباشرة أبدًا**. هذا تطبيق تعليمي.

## 3. مُنفّذ الأدوات

```python
TOOL_FUNCTIONS = {
    "lookup_word": lookup_word,
    "get_weather": get_weather,
    "calculate":  calculate,
}


def execute_tool(name: str, arguments: dict) -> str:
    """ابحث عن الأداة المناسبة ونفّذها مع التقاط الأخطاء."""
    if name not in TOOL_FUNCTIONS:
        return json.dumps({"error": f"أداة غير معروفة: {name}"})
    try:
        return TOOL_FUNCTIONS[name](**arguments)
    except Exception as e:
        return json.dumps({"error": f"فشل تنفيذ {name}: {e}"})
```

## 4. حلقة المحادثة

```python
def chat(user_message: str, history: list[dict] | None = None) -> str:
    """محادثة تدعم tool calling."""
    client = OpenAI(api_key=os.environ["OPENAI_API_KEY"])
    model = os.environ.get("OPENAI_MODEL", "gpt-4o-mini")

    if history is None:
        history = []

    history.append({"role": "user", "content": user_message})

    # 1. الطلب الأول
    response = client.responses.create(
        model=model,
        input=history,
        tools=tools,
    )

    # 2. هل طلب النموذج استدعاء أدوات؟
    tool_calls = [
        item for item in response.output
        if getattr(item, "type", None) == "function_call"
    ]

    if not tool_calls:
        # لا استدعاء. أضف الردّ للـ history وأعده.
        text = response.output_text
        history.append({"role": "assistant", "content": text})
        return text

    # 3. نفّذ كل استدعاء
    for call in tool_calls:
        args = json.loads(call.arguments)
        result = execute_tool(call.name, args)

        # أضف النتيجة لـ history
        history.append({
            "role": "tool",
            "tool_call_id": call.call_id,
            "content": result,
        })

    # 4. الطلب الثاني: مع النتائج
    response = client.responses.create(
        model=model,
        input=history,
        tools=tools,
    )

    text = response.output_text
    history.append({"role": "assistant", "content": text})
    return text
```

## 5. اختبارات

```python
"""
tests/test_tool_calling.py

اختبارات أساسية للتأكد أنّ:
- الأدوات تُعرَّف بشكل صحيح.
- تنفيذ الأدوات يعمل مع مدخلات سليمة.
- validation يرفض مدخلات خاطئة.
"""
import pytest
from my_app.tools import execute_tool, lookup_word, get_weather, calculate


def test_lookup_word_known():
    result = json.loads(lookup_word("ذكاء"))
    assert "definition" in result
    assert result["word"] == "ذكاء"


def test_lookup_word_unknown():
    result = json.loads(lookup_word("كلمة_غير_موجودة"))
    assert "error" in result


def test_get_weather_known():
    result = json.loads(get_weather("عمّان"))
    assert "temp_c" in result


def test_calculate_simple():
    result = json.loads(calculate("2 + 2"))
    assert result["result"] == 4


def test_calculate_blocks_dangerous():
    """تعبير يحاول الوصول لـ __import__ يجب أن يُرفض."""
    result = json.loads(calculate("__import__('os').system('rm -rf /')"))
    assert "error" in result


def test_execute_tool_unknown():
    result = json.loads(execute_tool("unknown_tool", {}))
    assert "error" in result
```


## أخطاء شائعة

- **"عدم فصل tools عن التنفيذ"**: ضع أدواتك في dict واضح. لا تبعثرها في الكود.
- **"eval(input) آمن"**: **ليس آمنًا أبدًا**. استخدم compile + whitelist كما في `calculate`.
- **"تنفيذ كل ما يطلبه النموذج"**: لا. validation وwhitelist.
- **"description غامض"**: النموذج لن يعرف متى يستدعي الأداة. كن محدّدًا.
- **"تسريب keys في logs"**: لا تطبع arguments كاملة، فقط المفاتيح.
