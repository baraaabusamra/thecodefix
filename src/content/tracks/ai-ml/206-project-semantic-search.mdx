## الهدف

نبني **نظام بحث دلالي** يجيب عن أسئلة بالعربية من مجموعة مستندات (مقالات، وثائق)، **بدون LLM وبدون قاعدة متجهات**. فقط embeddings + NumPy.

```text
input:  "ما الفرق بين HNSW و IVFFlat؟"
output: top-5 chunks مرتّبة بالتشابه، مع distances.
```

## المرحلة 2: Ingestion

```python
# src/ingest.py
from pathlib import Path
import numpy as np
from sentence_transformers import SentenceTransformer

MODEL_NAME = "intfloat/multilingual-e5-small"
DATA_DIR = Path("data/docs")
INDEX_PATH = Path("index/vectors.npz")

model = SentenceTransformer(MODEL_NAME)

def chunk_text(text: str) -> list[str]:
    """قسّم على فقرات (أسطر فارغة)."""
    return [p.strip() for p in text.split("\n\n") if p.strip()]

def ingest_all():
    chunks: list[str] = []
    sources: list[str] = []
    for doc_path in sorted(DATA_DIR.glob("*.txt")):
        text = doc_path.read_text(encoding="utf-8")
        for chunk in chunk_text(text):
            chunks.append(chunk)
            sources.append(doc_path.name)

    print(f"Total chunks: {len(chunks)}")

    # embeddings — multilingual-e5 يتطلّب prefix "passage:" للمستندات
    # (أو model.encode_document الذي يضيف نفس الـ prefix داخليًا).
    passages = [f"passage: {c}" for c in chunks]
    vectors = model.encode(
        passages,
        normalize_embeddings=True,
        show_progress_bar=True,
    )

    # حفظ
    INDEX_PATH.parent.mkdir(parents=True, exist_ok=True)
    np.savez(
        INDEX_PATH,
        vectors=vectors,
        chunks=np.array(chunks, dtype=object),
        sources=np.array(sources, dtype=object),
    )

    print(f"Saved index → {INDEX_PATH}")

if __name__ == "__main__":
    ingest_all()
```

> **ملاحظة**: المتجهات normalized → cosine = dot product (درس 121).

## المرحلة 3: Search

```python
# src/search.py
import sys
from pathlib import Path
import numpy as np
from sentence_transformers import SentenceTransformer

MODEL_NAME = "intfloat/multilingual-e5-small"
INDEX_PATH = Path("index/vectors.npz")

model = SentenceTransformer(MODEL_NAME)

def load_index():
    data = np.load(INDEX_PATH, allow_pickle=True)
    return data["vectors"], list(data["chunks"]), list(data["sources"])

def search(query: str, vectors: np.ndarray, chunks: list[str], sources: list[str], k: int = 5):
    # multilingual-e5 يتطلّب prefix "query:" للاستعلام (توثيق الموديل).
    q_vec = model.encode(
        f"query: {query}",
        normalize_embeddings=True,
    )

    # cosine = dot product عند normalized
    scores = vectors @ q_vec
    idx = np.argsort(-scores)[:k]

    results = []
    for i in idx:
        results.append({
            "score":   float(scores[i]),
            "chunk":   chunks[i],
            "source":  sources[i],
        })
    return results

def main():
    vectors, chunks, sources = load_index()
    print(f"Index: {len(chunks)} chunks\n")

    if len(sys.argv) > 1:
        queries = [" ".join(sys.argv[1:])]
    else:
        print("Enter query (empty to quit):")
        queries = [input("> ").strip()]
        while queries[-1]:
            queries.append(input("> ").strip())
        queries = queries[:-1]

    for q in queries:
        print(f"\nQ: {q}\n")
        results = search(q, vectors, chunks, sources, k=5)
        for i, r in enumerate(results, 1):
            print(f"[{i}] score={r['score']:.3f}  source={r['source']}")
            print(f"    {r['chunk'][:150]}...")
            print()

if __name__ == "__main__":
    main()
```

## المرحلة 4: التقييم

```python
# src/eval.py
"""تقييم recall@5 بسيط."""
import json
from pathlib import Path
from search import load_index, search, model  # noqa

EVAL_FILE = Path("eval_set.json")

def main():
    vectors, chunks, sources = load_index()
    eval_set = json.loads(EVAL_FILE.read_text(encoding="utf-8"))

    # eval_set: [{"question": "...", "relevant_sources": ["03-pgvector.txt", ...]}, ...]
    total_recall = 0.0
    total = len(eval_set)

    for item in eval_set:
        q = item["question"]
        relevant = set(item["relevant_sources"])
        results = search(q, vectors, chunks, sources, k=5)
        retrieved_sources = {r["source"] for r in results}
        hits = len(retrieved_sources & relevant)
        recall = hits / max(len(relevant), 1)
        total_recall += recall
        print(f"Q: {q[:60]}... → recall@5={recall:.2f}")

    print(f"\nMean recall@5: {total_recall / total:.3f}")

if __name__ == "__main__":
    main()
```

  },

    "question": "كيف يعمل RAG؟",
    "relevant_sources": ["01-rag-intro.txt", "02-embeddings.txt"]
  }
]

## الأخطاء الشائعة

- **استخدام model مختلف للـ index والـ query**: لا. نفس النموذج بالضبط.
- **تطبيع غير متّسق**: لا. normalize_embeddings=True في الـ index والاستعلام.
- **eval set صغير (3 أسئلة)**: لا. ≥ 10 أسئلة لإشارة ذات معنى.
- **عدم فصل relevant_sources عن chunks**: relevant_sources = filenames، لا chunk indices.
